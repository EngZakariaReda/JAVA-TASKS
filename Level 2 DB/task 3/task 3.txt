1. Find employees whose employee ID is between 100 and 105    SELECT *    FROM employees    WHERE employee_id BETWEEN 100 AND 105;2. Find employees whose IDs are in a specific set    SELECT *    FROM employees    WHERE employee_id IN (151, 152, 153, 154, 155);3. Find employees whose first name starts with P or p    SELECT *    FROM employees    WHERE UPPER(first_name) LIKE 'P%';4. Find employees whose first name ends with A or a    SELECT *    FROM employees    WHERE UPPER(first_name) LIKE '%A';5. Find employees whose first name contains A or a anywhere    SELECT *    FROM employees    WHERE UPPER(first_name) LIKE '%A%';6. Find employees whose third name character is E or e    SELECT *    FROM employees    WHERE UPPER(SUBSTR(first_name, 3, 1)) = 'E';7. Find employees who do not have a manager    SELECT *    FROM employees    WHERE manager_id IS NULL;8.
