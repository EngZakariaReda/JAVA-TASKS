1. Create Manger table

CREATE TABLE Manger (
	id		NUMBER(3), 
	name		varchar2(15), 
	age		NUMBER(3), 
	birth_date		date, 
	address		varchar2(20)
)



2. Drop address column

ALTER TABLE manger DROP COLUMN address ;



3. Add city_address and street columns

ALTER TABLE manger ADD (city_address varchar2(10), street varchar2(10))



4. Rename name column to full_name

ALTER TABLE manger RENAME column  name to full_name ;



5. Make Manger table READ ONLY

ALTER TABLE manger READ only ; 


6. Create Owner table from Manger -- Only: id, name, birth_date -- No data will be copied

CREATE TABLE owner (id , name , birth_date) AS SELECT id , full_name , birth_date
