-- Q1. Display current date in the format 'DD-MON-YYYY'.

SELECT TO_CHAR(SYSDATE, 'DD-MON-YYYY')
FROM DUAL;


-- Q2. Display current date in the format 'Month YYYY'.

SELECT TO_CHAR(SYSDATE, 'Month YYYY')
FROM DUAL;


-- Q3. Convert a number to character using TO_CHAR with formatting.

SELECT TO_CHAR(12345.67, '99,999.99')
FROM DUAL;


-- Q4. Show salary formatted with a currency symbol.

SELECT TO_CHAR(Salary, '$99,999.99')
FROM Employees;
