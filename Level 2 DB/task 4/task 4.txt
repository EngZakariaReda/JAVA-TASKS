CREATE TABLE STUDENTS (
    name VARCHAR2(50),
    marks NUMBER(3)
);

-- Q3. Use CASE to assign grades.

SELECT 
    name,
    marks,
    CASE
        WHEN marks >= 90 THEN 'A'
        WHEN marks >= 80 THEN 'B'
        WHEN marks >= 70 THEN 'C'
        ELSE 'F'
    END AS grade
FROM STUDENTS;
