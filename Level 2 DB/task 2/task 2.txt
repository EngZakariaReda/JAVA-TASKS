1. Create the Employees table

CREATE TABLE Employees (
    EmployeeID NUMBER PRIMARY KEY,
    FirstName VARCHAR2(50),
    LastName VARCHAR2(50),
    Department VARCHAR2(50),
    Salary NUMBER(10, 2)
);

3. Update the salary of employee 101 to 600000

UPDATE Employees
SET Salary = 600000
WHERE EmployeeID = 101;


4. Delete an employee whose id = 101

DELETE FROM Employees
WHERE EmployeeID = 101;


5. Retrieve all employees in the IT department

SELECT *
FROM Employees
WHERE Department = 'IT';


6. Display all employee data with FirstName and LastName concatenated into one column

SELECT EmployeeID,
FirstName || ' ' || LastName AS FullName,
Department,
Salary
FROM Employees;
